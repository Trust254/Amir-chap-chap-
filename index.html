function handleAuth(event){ event.preventDefault(); const phone = document .getElementById("authPhone") .value .trim(); const name = document .getElementById("authName") .value .trim(); if(!phone){ showToast("Please enter your phone number."); return; } // Get all registered users let registeredUsers = JSON.parse( localStorage.getItem("amirRegisteredUsers") ) || []; // ========================= // REGISTER // ========================= if(authMode === "register"){ if(!name){ showToast("Please enter your full name."); return; } // Check if phone already exists const existingUser = registeredUsers.find( user => user.phone === phone ); if(existingUser){ showToast( "This phone number is already registered. Please login." ); switchAuth("login"); document.getElementById( "authPhone" ).value = phone; return; } // Create new account const newUser = { id: "USR-" + Date.now(), name: name, phone: phone, createdAt: new Date().toISOString() }; registeredUsers.push(newUser); localStorage.setItem( "amirRegisteredUsers", JSON.stringify(registeredUsers) ); // Automatically login after registration currentUser = newUser; localStorage.setItem( "amirUser", JSON.stringify(currentUser) ); updateAuthUI(); document.getElementById( "authSection" ).style.display = "none"; showToast( "Account created successfully. Welcome " + currentUser.name + "!" ); return; } // ========================= // LOGIN // ========================= const existingUser = registeredUsers.find( user => user.phone === phone ); if(!existingUser){ showToast( "Account not found. Please register first." ); switchAuth("register"); document.getElementById( "authPhone" ).value = phone; return; } // Login existing account currentUser = existingUser; localStorage.setItem( "amirUser", JSON.stringify(currentUser) ); updateAuthUI(); document.getElementById( "authSection" ).style.display = "none"; showToast( "Welcome back, " + currentUser.name + "!" ); }
